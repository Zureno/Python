from aws_lambda_powertools import Logger
import os
import logging

# Detect environment
is_local = os.getenv('AWS_EXECUTION_ENV') is None

if is_local:
    # Completely suppress logger output locally
    # Only UCAS colored output will show
    logger = Logger(service="akamaivalidator-lambda", level="CRITICAL")
    
    # Also suppress boto3/botocore logs
    logging.getLogger('boto3').setLevel(logging.CRITICAL)
    logging.getLogger('botocore').setLevel(logging.CRITICAL)
    logging.getLogger('urllib3').setLevel(logging.CRITICAL)
else:
    # In Lambda: full logging
    logger = Logger(service="akamaivalidator-lambda", level="INFO")
