            $("#subPlanCodeId").val(subPlanCodeId);
            const frm = document.getElementById("frmDeleteJurisdiction");
            frm.action = "/gettaxcalcbac/PlanSubCodes/DeleteJurisdiction";
            frm.method = "POST";
            frm.submit();
        }

        function showJurisdictionsPopup(selectedId) {
            $('#selectedId').val(selectedId);
            const frm = document.getElementById("frmWithTokenPlanSubCodes");
            $.ajax({
                type: "POST",
                url: "/gettaxcalcbac/PlanSubCodes/AddDeleteJurisdiction",
                data: $(frm).serializeArray(),
                success: function (responseHtml) {
                    containerJurisdictions.html(responseHtml);
                    setUpPopupWindows();
                    showPopupByIdMiddle(JurisdictionsListTitle, 'popDivJurisdictionsListPlanSubCode');
                },
                error: function (data) {
                }
            });
        }

        function handleSelectAll(event) {
            var checking = event.target.checked;
            if (checking === false) {
                $(".selectAllCheckBox").prop("checked", false);
                $(".selectAllCheckBox").prop("title", "Select all");
            }
            else if ($('.selectCheckBox:checked').length == $('.selectCheckBox').length) {
                $(".selectAllCheckBox").prop("checked", true);
                $(".selectAllCheckBox").prop("title", "Unselect all");
            }
        }

        function genericSelectAll() {
            var checked = $(".selectAllCheckBox").prop("checked");
            $(".selectCheckBox").prop("checked", checked);
            if (checked === true) {
                $(".selectAllCheckBox").prop("title", "Unselect all");
            }
            else {
                $(".selectAllCheckBox").prop("title", "Select all");
            }
            return checked;
        }
    </script>
